cmake_minimum_required(VERSION 3.15)
project(universe_engine CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -march=native -mavx2 -mfma -flto -DNDEBUG -ffast-math -funroll-loops")
set(CMAKE_EXE_LINKER_FLAGS_RELEASE "-flto")
add_executable(universe_engine src/main.cpp src/indicators.cpp src/strategy.cpp src/optimizer.cpp)
target_include_directories(universe_engine PRIVATE src)
find_package(Threads REQUIRED)
target_link_libraries(universe_engine Threads::Threads)
